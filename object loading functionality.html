// Smoothly transition color back to original const color = new THREE.Color().lerpColors( child.material.userData.originalColor, new THREE.Color(1, 0.3, 0), glowIntensity ); child.material.color.copy(color); } } // Create an explosion particle from disintegration function createExplosionParticle(position, velocity, color) { // Use existing particles from the simulation for the explosion effect const particleIndex = Math.floor(Math.random() * params.particleCount);
